cmake_minimum_required(VERSION 3.8)
project(navigation_manager)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "CMake build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
set(dependencies rclcpp rclcpp_action geometry_msgs nav_msgs std_msgs std_srvs
  actionlib_msgs rcl_interfaces builtin_interfaces tf2 tf2_ros tf2_geometry_msgs path_planner known_locations_interfaces)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "action/Navigate.action" DEPENDENCIES geometry_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(navigation_core STATIC src/navigation_manager.cpp src/execution_monitor.cpp src/recovery_manager.cpp)
target_include_directories(navigation_core PUBLIC include)
ament_target_dependencies(navigation_core ${dependencies})
target_link_libraries(navigation_core "${typesupport}")
add_executable(navigation_manager_node src/navigation_node.cpp)
ament_target_dependencies(navigation_manager_node ${dependencies})
target_link_libraries(navigation_manager_node navigation_core)
install(TARGETS navigation_core navigation_manager_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_navigation test/test_navigation.cpp)
  target_link_libraries(test_navigation navigation_core)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(action_integration test/test_action_integration.py TIMEOUT 240)
  ament_add_pytest_test(real_planner_integration test/test_action_integration.py TIMEOUT 120
    ENV "NAVIGATION_TEST_REAL_PLANNER=1" "NAVIGATION_TEST_DOMAIN_ID=175"
      "PYTEST_ADDOPTS=-k 'test_01 or test_02 or test_03'")
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
